      if (email && email.value) body.email = email.value;
      fetch('/api/scan', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(body)
      })
        .then(function (r) { return r.json().catch(function () { return {}; }); })
        .then(function (data) {
          result.textContent = data.message ||

        })
        .catch(function () {
